// Entry-page presentation only. Inspection, estimate and garage handlers stay shared.
export function createQuickEntry({showStep,askMechanic,saveName}) {
  const input=document.getElementById('damage-photos');
  const hero=document.querySelector('.hero');
  const formCard=document.getElementById('chat-history').closest('.card');
  document.body.classList.add('quick-entry');
  const upload=document.createElement('button');
  upload.id='quick-upload';upload.className='cta';upload.type='button';
  upload.textContent=({en:'Upload photos of the damage',hi:'नुकसान की तस्वीरें अपलोड करें',mr:'नुकसानीचे फोटो अपलोड करा'})[window.i18n.language] || 'Upload photos of the damage';
  document.getElementById('headline').after(upload);
  upload.addEventListener('click',()=>input.click());
  showStep('quick-upload');formCard.hidden=true;
  let nameCollected=false;
  const estimateShown=()=>{
    const result=document.getElementById('result');
    if(!result.querySelector('.result-wrap') || result.querySelector('#quick-name-form'))return;
    if(nameCollected)return;
    const invitation=result.querySelector('.garage-invite');if(invitation)invitation.hidden=true;
    const card=document.createElement('section');card.className='card';
    const form=document.createElement('form');form.id='quick-name-form';
    const label=document.createElement('label');label.htmlFor='quick-name';label.textContent=window.i18n.t('Your name');
    const name=document.createElement('input');name.id='quick-name';name.type='text';name.autocomplete='name';name.maxLength=100;name.required=true;
    const submit=document.createElement('button');submit.className='cta';submit.type='submit';submit.textContent=window.i18n.t('Continue');
    form.append(label,name,submit);card.append(form);result.querySelector('.result-wrap').after(card);
    form.addEventListener('submit',event=>{event.preventDefault();if(!name.value.trim()){name.focus();return;}saveName(name.value.trim());
      const quoteName=result.querySelector('[data-quote-form] [name=name]');if(quoteName)quoteName.value=name.value.trim();
      nameCollected=true;window.trackAnalytics('chat_name_given');card.remove();if(invitation)invitation.hidden=false;});
  };
  new MutationObserver(estimateShown).observe(document.getElementById('result'),{childList:true,subtree:true});
  return {async photosReady(){
    if(document.body.dataset.flowStage!=='quick-upload')return;
    showStep('car');await askMechanic('car','What is your car’s make, model and variant?');
  }};
}
